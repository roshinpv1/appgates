import time
from functools import wraps

def rate_limit(max_requests, period):
    def decorator(func):
        requests = {}
        @wraps(func)
        def wrapper(*args, **kwargs):
            now = int(time.time())
            if now - requests.get('timestamp', 0) >= period:
                requests['timestamp'] = now
                requests['count'] = 1
            elif requests['count'] < max_requests:
                requests['count'] += 1
            else:
                return "Rate limit exceeded"
            return func(*args, **kwargs)
        wrapper.timestamp = now
        return wrapper
    return decorator
